/************************************************************ * KHỞI TẠO HỆ THỐNG ************************************************************/ function setupAll() { const stations = [ 'TBA 500kV Di Linh','TBA 220kV Bảo Lộc','TBA 220kV Đức Trọng', 'TBA 220kV Phan Thiết','TBA 220kV Hàm Tân','TBA 220kV Vĩnh Hảo', 'TBA 500kV Vĩnh Tân','TBA 500kV Đắk Nông','TBA 500kV Thuận Nam', 'TBA 220kV Đắk Nông','TBA 220kV Tháp Chàm','TBA 220kV Phan Rí', 'Kho Đất Nông','Kho Lâm Đồng','Kho Di Linh' ]; SpreadsheetApp.getUi().alert('Đã khởi tạo hệ thống thành công!'); } function ensureSheet(name, headers, sample) { let sh = SS.getSheetByName(name); if (!sh) sh = SS.insertSheet(name); if (sh.getLastRow() === 0) { sh.getRange(1,1,1,headers.length).setValues([headers]) .setFontWeight('bold').setBackground('#e8eaf6'); if (sample && sample.length) { sh.getRange(2,1,sample.length,headers.length).setValues(sample); } } return sh; } /************************************************************ * TRIGGER: onEdit – tự động tính khi nhập ************************************************************/ function onEdit(e) { if (!e || !e.range) return; const sh = e.range.getSheet(); if (sh.getName() !== SHEETS.NHATKY) return; const row = e.range.getRow(); if (row < 2) return; const col = e.range.getColumn(); // Chỉ tính khi sửa các cột liên quan const triggerCols = [C.NGAY,C.GIO_BD,C.GIO_KT,C.DC_DI_TU,C.DC_DI_DEN,C.DC_VE_TU,C.DC_VE_DEN]; if (triggerCols.indexOf(col) >= 0) { computeRow(row); } } /************************************************************ * LÕI TÍNH TOÁN – BÓC TÁCH KHOẢNG THỜI GIAN ************************************************************/ function computeRow(rowIndex) { const sh = SS.getSheetByName(SHEETS.NHATKY); const rng = sh.getRange(rowIndex, 1, 1, HEADERS.length); const row = rng.getValues()[0]; const ngay = row[C.NGAY-1]; const gioBD = row[C.GIO_BD-1]; const gioKT = row[C.GIO_KT-1]; if (!ngay || !gioBD || !gioKT) return; const workDay = toDate(ngay); workDay.setHours(0,0,0,0); const dow = workDay.getDay(); const isWeekend = (dow === 0 || dow === 6); const isHoliday = isHolidayDate(workDay); // Chuyển giờ sang phút tuyệt đối kể từ 0:00 của ngày công tác // Ngày công tác kéo dài từ 6:00 hôm nay tới 6:00 hôm sau (tức [360, 1800] phút) const startAbs = timeToAbs(gioBD, false); let endAbs = timeToAbs(gioKT, false); // Nếu giờ kết thúc < giờ bắt đầu => qua ngày hôm sau if (endAbs <= startAbs) endAbs += 1440; // Kẹp vào cửa sổ ngày công tác const ws = Math.max(startAbs, 360); const we = Math.min(endAbs, 1800); const res = { HC:0, OT_DAY:0, OT_NIGHT:0, WEEKEND_DAY:0, WEEKEND_NIGHT:0, HOLIDAY_DAY:0, HOLIDAY_NIGHT:0 }; if (we > ws) { splitAndClassify(ws, we, isWeekend, isHoliday, res); } const travelH = travelHours(row); const totalH = round(res.HC + res.OT_DAY + res.OT_NIGHT + res.WEEKEND_DAY + res.WEEKEND_NIGHT + res.HOLIDAY_DAY + res.HOLIDAY_NIGHT, 2); const weighted = round( res.HC*COEFF.HC + res.OT_DAY*COEFF.OT_DAY + res.OT_NIGHT*COEFF.OT_NIGHT + res.WEEKEND_DAY*COEFF.WEEKEND_DAY + res.WEEKEND_NIGHT*COEFF.WEEKEND_NIGHT + res.HOLIDAY_DAY*COEFF.HOLIDAY_DAY + res.HOLIDAY_NIGHT*COEFF.HOLIDAY_NIGHT + travelH*COEFF.TRAVEL, 2); } /** * Chia [startAbs, endAbs] thành các đoạn nhỏ theo ranh giới: * 360 (6:00) | 420 (7:00) | 690 (11:30) | 810 (13:30) | 1020 (17:00) | 1320 (22:00) | 1800 (6:00 hôm sau) * Rồi phân loại từng đoạn theo HC / OT / cuối tuần / lễ. */ function splitAndClassify(startAbs, endAbs, isWeekend, isHoliday, res) { const bounds = [360, 420, 690, 810, 1020, 1320, 1800]; const pts = {}; pts[startAbs] = 1; pts[endAbs] = 1; bounds.forEach(b => { if (b > startAbs && b < endAbs) pts[b] = 1; }); const sorted = Object.keys(pts).map(Number).sort((a,b)=>a-b); for (let i = 0; i < sorted.length - 1; i++) { const s = sorted[i], e = sorted[i+1]; const hours = (e - s) / 60; const mid = (s + e) / 2; const isDay = (mid >= 360 && mid < 1320); // 6:00–22:00 const inHC = (mid >= 420 && mid < 690) || (mid >= 810 && mid < 1020); if (isHoliday) { if (isDay) res.HOLIDAY_DAY += hours; else res.HOLIDAY_NIGHT += hours; } else if (isWeekend) { if (isDay) res.WEEKEND_DAY += hours; else res.WEEKEND_NIGHT += hours; } else if (inHC) { res.HC += hours; } else if (isDay) { res.OT_DAY += hours; } else { res.OT_NIGHT += hours; } } } /** Đổi giá trị ô (Date hoặc chuỗi 'HH:mm') thành phút kể từ 0:00 */ function timeToAbs(v, plusIfNight) { if (!v && v !== 0) return 0; let h, m; if (v instanceof Date) { h = v.getHours(); m = v.getMinutes(); } else { const s = String(v).trim(); const parts = s.split(':'); h = parseInt(parts[0], 10) || 0; m = parseInt(parts[1], 10) || 0; } return h * 60 + m; } /** Tính tổng giờ di chuyển (đi + về) – hệ số 1 */ function travelHours(row) { let total = 0; const pairs = [ [row[C.DC_DI_TU-1], row[C.DC_DI_DEN-1]], [row[C.DC_VE_TU-1], row[C.DC_VE_DEN-1]] ]; pairs.forEach(([a,b]) => { if (a == null || a === '' || b == null || b === '') return; let x = timeToAbs(a), y = timeToAbs(b); if (y <= x) y += 1440; total += (y - x) / 60; }); return total; } function isHolidayDate(d) { const sh = SS.getSheetByName(SHEETS.NGAYLE); if (!sh || sh.getLastRow() < 2) return false; const vals = sh.getRange(2,1, sh.getLastRow()-1, 1).getValues(); const key = Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); return vals.some(r => r[0] && Utilities.formatDate(toDate(r[0]), Session.getScriptTimeZone(), 'yyyy-MM-dd') === key); } function toDate(v) { if (v instanceof Date) return new Date(v); const parts = String(v).split(/[\/\-]/); if (parts.length === 3) { // dd/mm/yyyy if (parts[0].length <= 2 && parts[2].length === 4) { return new Date(+parts[2], +parts[1]-1, +parts[0]); } // yyyy-mm-dd if (parts[0].length === 4) { return new Date(+parts[0], +parts[1]-1, +parts[2]); } } return new Date(v); } function round(x, d) { const k = Math.pow(10, d); return Math.round(x*k)/k; } /************************************************************ * TỔNG HỢP THEO THÁNG / NGƯỜI / LOẠI CÔNG TÁC ************************************************************/ function buildSummary(params) { const sh = SS.getSheetByName(SHEETS.NHATKY); if (types && types.length && types.indexOf(r[C.LOAI_CT-1]) < 0) return false; return true; }); // Group: theo nhân viên + ngày → cộng dồn các giờ const map = {}; // key = manv|yyyy-mm-dd filtered.forEach(r => { const d = toDate(r[C.NGAY-1]); '; ':'') + r[C.GHI_CHU-1]; }); // Chuyển thành mảng, tính công SCL / ĐTXD (làm tròn >4h = 1 công) const rows = Object.values(map).map(o => { let congSCL = 0, congDTXD = 0, gioSCL = 0, gioDTXD = 0; Object.keys(o.details).forEach(code => { const h = o.details[code];
